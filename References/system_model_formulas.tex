\section{Communication Model}
% Source: [Xie] Eq.(1)--(9)
% =============================================================================

\subsection{Path Loss: LoS and NLoS}

The average path loss (dB) between UAV $l$ and user $k$ is computed as a
weighted sum of LoS and NLoS components \cite{Xie}:

\begin{equation}
  g_{l,k} = P_{l,k}^{\mathrm{LoS}} \cdot g_{l,k}^{\mathrm{LoS}}
           + \left(1 - P_{l,k}^{\mathrm{LoS}}\right) \cdot g_{l,k}^{\mathrm{NLoS}}
  \tag{Xie~1--2}
\end{equation}

\noindent where the LoS and NLoS path loss terms are:
\begin{align}
  g_{l,k}^{\mathrm{LoS}}  &= 20\log_{10}\!\left(\tfrac{4\pi f_c d_0}{c}\right)
                              + 10\,n_{\mathrm{LoS}}\log_{10}(d_{l,k}) + s_{\mathrm{LoS}}
  \tag{Xie~1} \\
  g_{l,k}^{\mathrm{NLoS}} &= 20\log_{10}\!\left(\tfrac{4\pi f_c d_0}{c}\right)
                              + 10\,n_{\mathrm{NLoS}}\log_{10}(d_{l,k}) + s_{\mathrm{NLoS}}
  \tag{Xie~2}
\end{align}

\noindent \textbf{Variables:}
$f_c$ = carrier frequency;
$d_0$ = reference distance;
$c$ = speed of light;
$n_{\mathrm{LoS}}, n_{\mathrm{NLoS}}$ = path loss exponents;
$s_{\mathrm{LoS}}, s_{\mathrm{NLoS}}$ = shadowing factors (dB);
$d_{l,k} = \sqrt{(X_l-X_k)^2+(Y_l-Y_k)^2+H^2}$ = 3D distance UAV--user.

\subsection{LoS Probability}

\begin{equation}
  P_{l,k}^{\mathrm{LoS}} = \frac{1}{1 + \kappa \exp\!\left(-\zeta\,(\theta_{l,k} - \kappa)\right)}
  \tag{Xie~3}
\end{equation}

\noindent where $\theta_{l,k} = \arcsin(H / d_{l,k})$ is the elevation angle (degrees),
and $\kappa,\,\zeta$ are environment-dependent constants.

\subsection{SINR: UAV $\to$ User (Downlink)}

\begin{equation}
  \mathrm{SINR}_{l,k} =
    \frac{P_{\mathrm{UAV}} \cdot 10^{-g_{l,k}/10}}
         {\displaystyle\sum_{l' \in \mathcal{L},\, l' \neq l}
          P_{\mathrm{UAV}} \cdot 10^{-g_{l',k}/10} + \sigma^2}
  \tag{Xie~4}
\end{equation}

\noindent \textbf{Variables:}
$P_{\mathrm{UAV}}$ = UAV transmit power (W);
$\sigma^2$ = noise power;
interference from all other UAVs $l' \neq l$.

\subsection{Downlink Rate: UAV $\to$ User}

\begin{equation}
  r_{l,k} = \frac{B}{M} \log_2\!\left(1 + \mathrm{SINR}_{l,k}\right)
  \tag{Xie~5}
\end{equation}

\noindent \textbf{Variables:}
$B$ = total bandwidth;
$M$ = number of users / sub-channels per UAV.

\subsection{Backhaul: MBS $\to$ UAV [Xie Eq.~(6)--(9)]}

\noindent In \cite{Xie}, the LoS and NLoS path losses for the wireless backhaul link (notation as in the paper) are
\begin{align}
  g_{\mathrm{BS},l}^{\mathrm{LoS}}  &= d_{\mathrm{BS},l}^{-\gamma},
  \tag{Xie~6} \\
  g_{\mathrm{BS},l}^{\mathrm{NLoS}} &= \eta\, d_{\mathrm{BS},l}^{-\gamma}.
  \tag{Xie~7}
\end{align}
The average path loss is
$g_{\mathrm{BS},l} = P_{\mathrm{BS},l}^{\mathrm{LoS}}\, g_{\mathrm{BS},l}^{\mathrm{LoS}}
 + \bigl(1 - P_{\mathrm{BS},l}^{\mathrm{LoS}}\bigr)\, g_{\mathrm{BS},l}^{\mathrm{NLoS}}$
(see the text immediately after (7) in \cite{Xie}). The received SINR is
\begin{equation}
  \mathrm{SINR}_{\mathrm{BS},l} =
    \frac{P_{\mathrm{BS}}\, 10^{-g_{\mathrm{BS},l}/10}}{\sigma^2}
  \tag{Xie~8}
\end{equation}
and the backhaul rate is
\begin{equation}
  r_{\mathrm{BS},l} = \frac{B_h}{L} \log_2\!\left(1 + \mathrm{SINR}_{\mathrm{BS},l}\right).
  \tag{Xie~9}
\end{equation}

\noindent \textbf{Variables:}
$B_h$ = backhaul bandwidth;
$L$ = number of UAVs;
$P_{\mathrm{BS}}$ = MBS transmit power;
$\sigma^2$ = noise power;
$\gamma$, $\eta$ = backhaul path loss exponent and NLoS excess factor as in \cite{Xie}.

\noindent \textbf{Code note:} \texttt{models.\_rate\_backhaul} computes an equivalent weighted channel gain
(LoS probability $\times$ distance-dependent term) then $\mathrm{SINR}$ and $r_{\mathrm{BS},l}$; the structure matches
(8)--(9), while the split (6)/(7) is folded into one gain expression for implementation.

\section{Request indicator, popularity, and cache (Xie)}
% =============================================================================

\noindent \textbf{Theo \cite{Xie} (Section III-A2):} Ký hiệu $p_{k,f,z} \in \{0,1\}$ là \emph{chỉ báo}:
$p_{k,f,z}=1$ khi user $k$ đang yêu cầu chunk $v_{f,z}$, ngược lại bằng $0$.
Các yêu cầu chunk chịu ảnh hưởng độ phổ biến; trong bài báo,
\emph{``the popularity of video chunks follows a Zipf-like distribution''} \cite{Xie}
(tham chiếu [37] trong bài gốc). Các Eq.~(Xie~10)--(Xie~12) lấy tổng theo $(f,z)$ với các hệ số
$p_{k,f,z}$; với mỗi request thực tế chỉ một cặp $(f,z)$ có $p_{k,f,z}=1$, các hạng tử còn lại bằng $0$.

\noindent \textbf{Cài đặt mô phỏng (luận văn):} Mỗi bước cần \emph{một} cặp $(f,z)$ theo độ phổ biến Zipf-like.
Code dùng phân phối xác suất joint trên tất cả cặp $(f,z)$ với trọng số Zipf theo thứ hạng
$r=1,\ldots,FZ$ (\texttt{environment.\_compute\_zipf\_joint\_probs}), rồi lấy mẫu một cặp ---
tương đương việc sinh một mẫu chỉ báo $p_{k,f,z}$ trong mô phỏng Monte Carlo.
Chi tiết công thức Zipf trên từng cặp là \emph{lựa chọn implementation}; bài Xie chỉ nêu Zipf-like, không ghi đủ hệ số từng rank.

\subsection{Cache placement and capacity [Xie, constraint (15a)]}

\noindent Ma trận cache nhị phân $y_{l,f,z} \in \{0,1\}$ (UAV $l$ lưu chunk $f$ ở bitrate $z$).
Ràng buộc dung lượng:
\begin{equation}
  \sum_{f=1}^{F}\sum_{z=1}^{Z} y_{l,f,z}\, s_{f,z} \leq C_l^{\mathrm{cache}}, \quad \forall\, l \in \mathcal{L}.
  \tag{Xie~15a}
\end{equation}

\section{Video Delivery Delay Model}
% Source: [Xie] Eq.(10)--(14)
% =============================================================================

\subsection{Total Delivery Delay}

\begin{equation}
  D_{l,k} = D_{l,k}^1 + D_{l,k}^2 + D_{l,k}^3
  \tag{Xie~13}
\end{equation}

\subsection{Scenario 1 — Direct Cache Hit}
\emph{UAV $l$ caches content $f$ at exactly the requested bitrate $z$.}

\begin{equation}
  D_{l,k}^1 = \sum_{f=1}^{F}\sum_{z=1}^{Z}
    p_{k,f,z}\, y_{l,f,z}\, \frac{s_{f,z}}{r_{l,k}}
  \tag{Xie~10}
\end{equation}

\noindent \textbf{Variables (Xie):}
$p_{k,f,z} \in \{0,1\}$ = chỉ báo user $k$ có yêu cầu $v_{f,z}$ hay không;
$y_{l,f,z} \in \{0,1\}$ = cache indicator (UAV $l$, content $f$, bitrate $z$);
$s_{f,z}$ = chunk size at bitrate $z$;
$r_{l,k}$ = downlink rate from Eq.~(Xie~5).

\subsection{Scenario 2 — Transcoding Hit}
\emph{UAV $l$ has a higher-bitrate version $z^+$ and transcodes down to $z$.}

\begin{equation}
  D_{l,k}^2 = \sum_{f=1}^{F}\sum_{z=1}^{Z}
    p_{k,f,z}\,(1 - y_{l,f,z})\, h_{l,f,z}
    \left(\frac{s_{f,z}}{r_{l,k}} + \frac{w_0\,(s_{f,z^+} - s_{f,z})}{c_{l,k}}\right)
  \tag{Xie~11}
\end{equation}

\noindent \textbf{Variables:}
$h_{l,f,z} = \min\!\left(\sum_{z'>z} y_{l,f,z'},\,1\right)$ = indicator that a higher bitrate is cached;
$z^+$ = lowest cached bitrate above $z$;
$w_0$ = CPU cycles per bit for transcoding;
$c_{l,k} = C_{\mathrm{comp}} / M$ = computing resource allocated to user $k$.

\subsection{Scenario 3 — Cache Miss}
\emph{Content not available; UAV fetches from MBS backhaul.}

\begin{equation}
  D_{l,k}^3 = \sum_{f=1}^{F}\sum_{z=1}^{Z}
    p_{k,f,z}
    \left(1 - \sum_{z'=z}^{Z} y_{l,f,z'}\right)
    \left(\frac{s_{f,z}}{r_{l,k}} + \frac{s_{f,z}}{r_{\mathrm{BS},l}}\right)
  \tag{Xie~12}
\end{equation}

\noindent $r_{\mathrm{BS},l}$ = backhaul rate from Eq.~(Xie~9).

\subsection{ABR segment size and client buffer (thesis implementation)}
\label{sec:abr-impl}

Trong code hiện tại, các representation ABR dùng \emph{cùng} thời lượng segment
$T_{\mathrm{seg}}$ (giây) như MPEG-DASH; kích thước chunk (bit) thống nhất với
Eq.~(Xie~10)--(Xie~12) là
\begin{equation}
  s_{f,z} = R_z\, T_{\mathrm{seg}},
  \tag{impl-ABR-s}
\end{equation}
với $R_z$ bitrate nominal của representation $z$ (bit/s).
Tham số \texttt{abr\_segment\_duration\_s}, \texttt{abr\_bitrate\_values\_kbps} trong
\texttt{config.py}; hàm \texttt{models.\_chunk\_size\_bits} và
\texttt{models.abr\_bitrate\_kbps\_list} đảm bảo cùng bảng $R_z$ giữa độ trễ và cache.

Gọi $D$ là độ trễ phục vụ một request theo một trong ba kịch bản (Xie~10), (Xie~11) hoặc (Xie~12)
--- \texttt{calculate\_total\_cost} trong \texttt{models.py}. Buffer phía client (giây):
\begin{equation}
  B_{t+1} = \min\!\left(B_{\max},\, \max\!\left(0,\, B_t + T_{\mathrm{seg}} - D\right)\right).
  \tag{impl-buffer}
\end{equation}
Thời gian rebuffering trong bước là $\max(0,\, -(B_t + T_{\mathrm{seg}} - D))$ khi $B_t + T_{\mathrm{seg}} < D$.

\subsection{Total System Delay Objective}

\begin{equation}
  D_{\mathrm{tot}} = \sum_{l \in \mathcal{L}} \sum_{k \in \mathcal{K}}
    x_{l,k} \cdot D_{l,k}
  \tag{Xie~14}
\end{equation}

\noindent \textbf{Variables:}
$x_{l,k} \in \{0,1\}$ = user association indicator (user $k$ assigned to UAV $l$);
$\mathcal{L}$ = set of UAVs; $\mathcal{K}$ = set of users.

\section{Optimization Problem}
% Source: [Xie] Problem (15)
% =============================================================================

\subsection{Joint CA+UA Optimization (Xie et al.)}

\begin{equation}
  \min_{\mathbf{x},\,\mathbf{y}} \quad D_{\mathrm{tot}}
    = \sum_{l \in \mathcal{L}} \sum_{k \in \mathcal{K}} x_{l,k} \cdot D_{l,k}
  \tag{Xie~P}
\end{equation}
\begin{align}
  \text{s.t.}\quad
  & \sum_{l \in \mathcal{L}} x_{l,k} = 1, \quad \forall\, k \in \mathcal{K}
    \tag{15b} \\
  & \sum_{k \in \mathcal{K}} x_{l,k} \leq M, \quad \forall\, l \in \mathcal{L}
    \tag{15c} \\
  & \sum_{f,z} y_{l,f,z} \cdot s_{f,z} \leq C_l^{\mathrm{cache}}, \quad \forall\, l
    \tag{15a} \\
  & x_{l,k},\, y_{l,f,z} \in \{0,1\}
    \tag{15d}
\end{align}

\noindent \textbf{Ghi chú:} Luận văn dùng D3QN để học chính sách; hàm mục tiêu lý thuyết
trong \cite{Xie} là tối thiểu hóa độ trễ $D_{\mathrm{tot}}$; phần thưởng thực tế là QoE ABR
(Mục~\ref{sec:rl-reward-impl}).

\section{D3QN và cài đặt luận văn}
% Baseline QEA: [Xie] Algorithm~1; D3QN: mô tả chuẩn (Double DQN + Dueling), không trích số hiệu từ bài khác.
% =============================================================================

\subsection{Q-Value Update (Double DQN)}

\begin{align}
  Q_1(s_t, a_t) &\leftarrow Q_1(s_t,a_t) + \alpha\!\left(
    r_{t+1} + \gamma\, Q_2\!\left(s_{t+1}, \arg\max_{a'} Q_1(s_{t+1},a')\right)
    - Q_1(s_t,a_t)\right)
  \tag{D3QN-a} \\
  Q_2(s_t, a_t) &\leftarrow Q_2(s_t,a_t) + \alpha\!\left(
    r_{t+1} + \gamma\, Q_1\!\left(s_{t+1}, \arg\max_{a'} Q_2(s_{t+1},a')\right)
    - Q_2(s_t,a_t)\right)
  \tag{D3QN-b}
\end{align}

\subsection{Dueling Network Q-Value Decomposition}

\begin{equation}
  Q(s,a) = V(s) + \left( A(s,a) - \frac{1}{|\mathcal{A}|} \sum_{a'} A(s,a') \right)
  \tag{D3QN-duel}
\end{equation}

\noindent \textbf{Variables:}
$V(s)$ = state value function;
$A(s,a)$ = advantage function;
normalization term ensures identifiability.

\subsection{Action Selection}

\begin{equation}
  a_t = \arg\max_{a}
    \left( V(s_t) + A(s_t,a) - \frac{1}{|\mathcal{A}|}\sum_{a'} A(s_t,a') \right)
  \tag{D3QN-act}
\end{equation}

\subsection{Target Value (D3QN)}

\begin{equation}
  Q_{\mathrm{target}} = R_{t+1} + \gamma\,\max_{A}\,Q(S_{t+1}, A;\,\theta_t^-)
  \tag{D3QN-tgt}
\end{equation}

\subsection{Reward trong code (QoE ABR)}
\label{sec:rl-reward-impl}

\noindent \texttt{environment.py} dùng phần thưởng QoE kiểu Pensieve cho ABR
(gắn với độ trễ $D$ từ mô hình Xie ở trên):
\begin{equation}
  r_t = u(R_{z_t}) - \beta_{\mathrm{rebuf}}\, t_{\mathrm{rebuf},t}
        - \beta_{\mathrm{sw}}\, \bigl| R_{z_t} - R_{z_{t-1}} \bigr|_{\mathrm{norm}},
  \tag{impl-QoE}
\end{equation}
trong đó $u(\cdot)$ là log hoặc tuyến tính theo bitrate nominal $R_{z_t}$;
$t_{\mathrm{rebuf},t}$ lấy từ cập nhật buffer (Mục~\ref{sec:abr-impl});
$\beta_{\mathrm{rebuf}}, \beta_{\mathrm{sw}}$ là \texttt{abr\_rebuffer\_penalty},
\texttt{abr\_switch\_penalty} trong \texttt{config.py}.

\subsection{Vector trạng thái trong code}
\label{sec:state-impl}

\noindent Vector trạng thái có số chiều cố định theo topology:
\begin{equation}
  |\mathbf{s}| = 2 + 7L + 2Z + 3,
  \tag{impl-state}
\end{equation}
với $L$ số UAV, $Z = |\{\text{bitrate levels}\}|$: vị trí xe và UAV (chuẩn hóa), khoảng cách xe--UAV,
độ đầy cache từng UAV, ba đặc trưng/UAV (cache hit tại $(f,z_{\mathrm{req}})$, tải, khả năng transcode),
trọng số joint Zipf $P(f_{\mathrm{req}},z_{\mathrm{req}})$ (không phải chỉ báo $p_{k,f,z}$ của \cite{Xie}),
one-hot $z_{\mathrm{req}}$, buffer chuẩn hóa, one-hot bitrate bước trước,
throughput EWMA chuẩn hóa. Chi tiết: \texttt{VanetEnvironment.get\_state} trong \texttt{environment.py}.

\subsection{Không gian hành động trong code (Xie, UAV-only)}

\noindent \textbf{Cài đặt luận văn (tầng UAV, Xie):}
$a \in \{0,\ldots, L Z \cdot 2 - 1\}$ với giải mã
$(\mathrm{uav\_idx}, z_{\mathrm{cached}}, \mathrm{cache\_dec})$; chỉ UAV (không MBS trong action);
xem \texttt{encode\_action} / \texttt{\_decode\_action} trong \texttt{environment.py}.

\section{QEA Baseline (for Comparison)}
% Source: [Xie] Algorithm 1--2, Eq.(21)--(24), Table 1

\subsection{Q-bit Observation}

\begin{equation}
  x_{l,k} = \begin{cases} 1 & \text{if } \mathrm{rand}() < |\beta_{l,k}|^2 \\ 0 & \text{otherwise} \end{cases}
  \tag{Xie~21}
\end{equation}

\begin{equation}
  y_{l,f,z} = \begin{cases} 1 & \text{if } \mathrm{rand}() < |\beta_{l,f,z}|^2 \\ 0 & \text{otherwise} \end{cases}
  \tag{Xie~22}
\end{equation}

\subsection{Rotation Gate Update}

\begin{equation}
  \begin{pmatrix} \alpha' \\ \beta' \end{pmatrix}
  = \begin{pmatrix} \cos\Delta\theta & -\sin\Delta\theta \\
                    \sin\Delta\theta &  \cos\Delta\theta \end{pmatrix}
  \begin{pmatrix} \alpha \\ \beta \end{pmatrix}
  \tag{Xie~23--24}
\end{equation}

\noindent $\Delta\theta$ is determined by the lookup Table~1 in \cite{Xie} based on
$(x_{\mathrm{cur}},\, b_{\mathrm{best}},\, f(\mathbf{X}_i) \geq f(\mathbf{B}))$.
